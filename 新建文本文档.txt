  <h1>AI 对话助手</h1>
  <div id="chat"></div>
  <input type="text" id="msg" placeholder="输入问题...">
  <button onclick="send()">发送</button>

<script>
async function send() {
  let msg = document.getElementById("msg").value;
  document.getElementById("chat").innerHTML += "你：" + msg + "<br>";

  // 这里调用免费 AI 接口
  let res = await fetch("https://api.52v.cn/api/ai?msg=" + encodeURIComponent(msg));
  let data = await res.json();

  document.getElementById("chat").innerHTML += "AI：" + data.content + "<br><br>";
  document.getElementById("msg").value = "";
}
</script>
